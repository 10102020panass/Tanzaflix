    const API_URL = 'http://localhost:5000/api';
    let selectedMethod = 'selcom-ussd';
    let movieId = null;
    let moviePrice = 0;
    let token = localStorage.getItem('token');

    // Check if user is logged in
    if (!token) {
      window.location.href = 'index.html';
    }

    // Get movie ID from URL
    const urlParams = new URLSearchParams(window.location.search);
    movieId = urlParams.get('movieId');

    if (!movieId) {
      showMessage('Hakuna movie iliyochaguliwa', 'error');
      setTimeout(() => window.location.href = 'main.html', 2000);
    }

    // Load movie info
    async function loadMovieInfo() {
      try {
        const response = await fetch(`${API_URL}/movies/${movieId}`, {
          headers: {
            'Authorization': `Bearer ${token}`
          }
        });

        if (!response.ok) throw new Error('Failed to load movie');

        const movie = await response.json();
        moviePrice = movie.price;

        // Display movie info
        document.getElementById('movieTitle').textContent = movie.title;
        document.getElementById('movieDesc').textContent = movie.description || 'Filamu nzuri';
        document.getElementById('movieYear').textContent = `Mwaka: ${movie.year} | Ubora: ${movie.quality}`;
        document.getElementById('movieImage').src = movie.image_url || 'assets/default-movie.jpg';
        document.getElementById('priceAmount').textContent = `${moviePrice.toLocaleString()} TZS`;
        document.getElementById('btnAmount').textContent = `${moviePrice.toLocaleString()} TZS`;
      } catch (error) {
        console.error('Error loading movie:', error);
        showMessage('Hitilafu katika kupakia filamu', 'error');
      }
    }

    // Payment method selection
    document.querySelectorAll('.method-card').forEach(card => {
      card.addEventListener('click', function() {
        document.querySelectorAll('.method-card').forEach(c => c.classList.remove('active'));
        this.classList.add('active');
        selectedMethod = this.dataset.method;
      });
    });

    // Payment form submission
    document.getElementById('paymentForm').addEventListener('submit', async (e) => {
      e.preventDefault();
      
      const phoneNumber = document.getElementById('phoneNumber').value.trim();

      if (!phoneNumber) {
        showMessage('Tafadhali ingiza nambari ya simu', 'error');
        return;
      }

      if (phoneNumber.length < 9) {
        showMessage('Nambari ya simu haupo salama', 'error');
        return;
      }

      // Show loading
      const spinner = document.getElementById('spinner');
      const btnText = document.getElementById('btnText');
      const btn = document.getElementById('paymentBtn');
      
      spinner.style.display = 'block';
      btn.disabled = true;

      try {
        const response = await fetch(`${API_URL}/payment/initiate`, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${token}`
          },
          body: JSON.stringify({
            movieId: parseInt(movieId),
            phoneNumber: phoneNumber,
            paymentMethod: selectedMethod
          })
        });

        const data = await response.json();

        if (!response.ok) {
          throw new Error(data.message || 'Failed to process payment');
        }

        showMessage(`Jukumu limepokelewa! Angalia simu yako kwa ujumbe wa USSD.`, 'success');
        
        // Save phone number
        localStorage.setItem('userPhone', phoneNumber);

        // Redirect after 3 seconds
        setTimeout(() => {
          window.location.href = `main.html?paymentStatus=processing&transactionId=${data.transactionId}`;
        }, 3000);

      } catch (error) {
        console.error('Payment error:', error);
        showMessage(error.message || 'Hitilafu katika kulipia', 'error');
      } finally {
        spinner.style.display = 'none';
        btn.disabled = false;
      }
    });
